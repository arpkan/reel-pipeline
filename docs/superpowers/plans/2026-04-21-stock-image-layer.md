# Free Stock Image Layer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a free stock photo search layer (Unsplash, Pexels, Pixabay) that tries to find a real photo before falling back to AI-generated Imagen images — for both reels and regular posts.

**Architecture:** A new `StockImageService` in `src/services/stock-image.ts` extracts a search query from the existing image prompt, searches all three free stock APIs in parallel, picks the best result, and returns a `Buffer`. The two image generation stages (`reel-stage3-images.ts` and `stage3b-image-gen.ts`) try stock first; if it returns `null`, they fall back to Imagen unchanged. Stock API keys are optional — if missing, the stock layer is silently skipped.

**Tech Stack:** Native `fetch` (Node >=22), Vitest, TypeScript ESM. No new npm packages needed.

```ts
// src/services/stock-image.ts
const STOPWORDS = [
  /\btext\s+reading\b/i,
  /\bno\s+(faces?|logos?|trademarks?|ui|people)\b/i,
  /\b(aspect\s+ratio|vertical|horizontal|square|portrait|landscape)\b/i,
  /\bsans-serif\b/i,
  /\bphotorealistic\b/i,
  /\bcinematic\b/i,
  /\bhigh\s+contrast\b/i,
  /\bno\s+(frames?|borders?)\b/i,
  /\bno\s+social\s+media\b/i,
];

export function extractSearchQuery(prompt: string): string {
  let cleaned = prompt;

  for (const re of STOPWORDS) {
    cleaned = cleaned.replace(re, "");
  }

  const firstComma = cleaned.indexOf(",");
  const firstPeriod = cleaned.indexOf(".");
  let raw: string;

  if (firstComma !== -1 && (firstPeriod === -1 || firstComma < firstPeriod)) {
    raw = cleaned.slice(0, firstComma).trim();
  } else if (firstPeriod !== -1) {
    raw = cleaned.slice(0, firstPeriod).trim();
  } else {
    raw = cleaned.trim();
  }

  raw = raw.replace(/^(A|An|The)\s+/i, "");

  if (raw.length > 100) {
    const truncated = raw.slice(0, 100);
    const lastSpace = truncated.lastIndexOf(" ");
    raw = truncated.slice(0, lastSpace > 0 ? lastSpace : 100);
  }

  return raw;
}
```

```ts
type Orientation = "portrait" | "square" | "landscape";

export async function searchUnsplash(
  query: string,
  orientation: Orientation,
  apiKey: string,
): Promise<string | null> {
  try {
    const params = new URLSearchParams({
      query,
      orientation,
      per_page: "3",
    });
    const res = await fetch(`https://api.unsplash.com/search/photos?${params}`, {
      headers: { Authorization: `Client-ID ${apiKey}` },
    });
    if (!res.ok) return null;
    const data = await res.json() as { results: Array<{ urls: { regular: string } }> };
    return data.results[0]?.urls.regular ?? null;
  } catch {
    return null;
  }
}

export async function searchPexels(
  query: string,
  orientation: Orientation,
  apiKey: string,
): Promise<string | null> {
  try {
    const params = new URLSearchParams({
      query,
      orientation,
      per_page: "3",
    });
    const res = await fetch(`https://api.pexels.com/v1/search?${params}`, {
      headers: { Authorization: apiKey },
    });
    if (!res.ok) return null;
    const data = await res.json() as { photos: Array<{ src: { large: string } }> };
    return data.photos[0]?.src.large ?? null;
  } catch {
    return null;
  }
}

export async function searchPixabay(
  query: string,
  orientation: Orientation,
  apiKey: string,
): Promise<string | null> {
  try {
    const params = new URLSearchParams({
      key: apiKey,
      q: query,
      image_type: "photo",
      orientation: orientation === "portrait" ? "vertical" : orientation === "landscape" ? "horizontal" : "all",
      per_page: "3",
    });
    const res = await fetch(`https://pixabay.com/api/?${params}`);
    if (!res.ok) return null;
    const data = await res.json() as { hits: Array<{ largeImageURL: string }> };
    return data.hits[0]?.largeImageURL ?? null;
  } catch {
    return null;
  }
}
```

```ts
export class StockImageService {
  private readonly unsplashKey: string;
  private readonly pexelsKey: string;
  private readonly pixabayKey: string;

  constructor(unsplashKey: string, pexelsKey: string, pixabayKey: string) {
    this.unsplashKey = unsplashKey;
    this.pexelsKey = pexelsKey;
    this.pixabayKey = pixabayKey;
  }

  async tryStockImage(
    prompt: string,
    orientation: "portrait" | "square" = "portrait",
  ): Promise<Buffer | null> {
    if (!this.unsplashKey && !this.pexelsKey && !this.pixabayKey) return null;

    const query = extractSearchQuery(prompt);

    const searchResults = await Promise.all([
      this.unsplashKey ? searchUnsplash(query, orientation, this.unsplashKey) : null,
      this.pexelsKey ? searchPexels(query, orientation, this.pexelsKey) : null,
      this.pixabayKey ? searchPixabay(query, orientation, this.pixabayKey) : null,
    ]);

    const imageUrl = searchResults.find((url): url is string => url !== null);
    if (!imageUrl) return null;

    try {
      const res = await fetch(imageUrl);
      if (!res.ok) return null;
      const arrayBuf = await res.arrayBuffer();
      return Buffer.from(arrayBuf);
    } catch {
      return null;
    }
  }
}

export function createStockImageService(): StockImageService {
  return new StockImageService(
    process.env.UNSPLASH_ACCESS_KEY ?? "",
    process.env.PEXELS_API_KEY ?? "",
    process.env.PIXABAY_API_KEY ?? "",
  );
}
```

```ts
import fs from "fs";
import path from "path";
import { spawnSync } from "child_process";
import type { ReelContext, ReelStage } from "../types.js";
import { createImageGenService, ImageSafetyError } from "../services/image-gen.js";
import { createStockImageService } from "../services/stock-image.js";

export class ReelStage3Images implements ReelStage {
  name = "reel-stage3-images";

  async run(ctx: ReelContext): Promise<Partial<ReelContext>> {
    if (!ctx.script) throw new Error("No script — run reel-stage1-script first");

    const imageGen = createImageGenService();
    const stockService = createStockImageService();
    const tmpDir = ctx.workDir!;
    fs.mkdirSync(tmpDir, { recursive: true });

    const imagePaths: string[] = [];

    for (const scene of ctx.script.scenes) {
      const outPath = path.join(tmpDir, `scene${scene.sceneNumber}.png`);
      console.log(`  Scene ${scene.sceneNumber}: ${scene.imagePrompt.slice(0, 70)}...`);

      let buffer: Buffer;

      const stockBuffer = await stockService.tryStockImage(scene.imagePrompt, "portrait");
      if (stockBuffer) {
        console.log(`    → stock image found`);
        buffer = stockBuffer;
      } else {
        try {
          buffer = await imageGen.generate(scene.imagePrompt, { aspectRatio: "9:16" });
        } catch (err) {
          if (err instanceof ImageSafetyError) {
            const safePrompt = `Cinematic news photo, no people visible: ${scene.imagePrompt.replace(/\b(child|kid|boy|girl|man|woman|person|people|minor|\d+-year-old)\b/gi, "").trim()}. Dramatic lighting, photorealistic.`;
            console.log(`    ⚠️ IMAGE_SAFETY — retrying with safe prompt`);
            buffer = await imageGen.generate(safePrompt, { aspectRatio: "9:16" });
          } else {
            throw err;
          }
        }
      }

      fs.writeFileSync(outPath, buffer);

      const probeResult = spawnSync("ffprobe", [
        "-v", "error", "-select_streams", "v:0",
        "-show_entries", "stream=width,height",
        "-of", "csv=p=0", outPath,
      ], { encoding: "utf8" });
      const [w, h] = probeResult.stdout.trim().split(",").map(Number);
      if (w && h && w >= h) {
        console.log(`    ⚠️ Landscape detected (${w}x${h}) — cropping to portrait`);
        const targetW = Math.floor(h * 9 / 16);
        const x = Math.floor((w - targetW) / 2);
        const croppedPath = outPath.replace(".png", "_portrait.png");
        spawnSync("ffmpeg", [
          "-i", outPath,
          "-vf", `crop=${targetW}:${h}:${x}:0`,
          "-y", croppedPath,
        ]);
        fs.renameSync(croppedPath, outPath);
      }

      imagePaths.push(outPath);
      const kb = (fs.statSync(outPath).size / 1024).toFixed(0);
      console.log(`    → ${outPath} (${kb} KB)`);
    }

    return { imagePaths };
  }
}
```

```ts
import type { Stage, RunContext, Services, Post } from "../types.js";
import { updatePost } from "../db/queries.js";

const MAX_ATTEMPTS = 3;

export class Stage3bImageGen implements Stage {
  name = "stage3b-image-gen";

  async run(ctx: RunContext, services: Services): Promise<Partial<RunContext>> {
    const posts: Post[] = [...ctx.posts];

    for (const post of posts) {
      if (post.imagePath && !post.imageFailed) {
        console.log(`[${ctx.runId}] skipping image gen for post ${post.index + 1} — already generated`);
        continue;
      }
      await this.generateForPost(post, ctx.runId, ctx.brand.gcsPrefix, services);
    }

    return { posts };
  }

  async generateForPost(
    post: Post,
    runId: string,
    gcsPrefix: string,
    services: Services
  ): Promise<void> {
    const filename = `post${post.index + 1}.png`;

    const stockBuffer = await services.stockImage.tryStockImage(post.imagePrompt, "square");
    if (stockBuffer) {
      console.log(`[${runId}] post ${post.index + 1}: using stock image`);
      const url = await services.gcs.uploadImage(gcsPrefix, runId, filename, stockBuffer);
      post.imagePath = url;
      post.imageFailed = false;
      updatePost(runId, post.index, { imagePath: url, imageFailed: false });
      return;
    }

    let lastError: string | undefined;

    for (let attempt = 1; attempt <= MAX_ATTEMPTS; attempt++) {
      try {
        const buffer = await services.imageGen.generate(post.imagePrompt);
        const url = await services.gcs.uploadImage(gcsPrefix, runId, filename, buffer);

        post.imagePath = url;
        post.imageFailed = false;
        updatePost(runId, post.index, { imagePath: url, imageFailed: false });
        return;
      } catch (err) {
        lastError = err instanceof Error ? err.message : String(err);
        console.warn(`[${runId}] image gen attempt ${attempt}/${MAX_ATTEMPTS} failed for post ${post.index + 1}: ${lastError}`);

        if (attempt < MAX_ATTEMPTS) {
          post.imagePrompt = post.imagePrompt + " Artistic illustration style.";
        }
      }
    }

    console.error(`[${runId}] image gen failed for post ${post.index + 1} after ${MAX_ATTEMPTS} attempts: ${lastError}`);
    post.imageFailed = true;
    post.imagePath = undefined;
    updatePost(runId, post.index, { imageFailed: true });
  }
}
```
